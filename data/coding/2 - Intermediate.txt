--- tests

word_counts("(wow) 'wow'") == {"wow": 2}

--- hint
Use text.lower().split(), then word.strip(".,!?;:\"'()") on each word, skipping any word that becomes empty. Use d[w] = d.get(w, 0) + 1 to count. For top_words, sort the items with key=lambda pair: (-pair[1], pair[0]) and slice the first k.
--- solution
def word_counts(text):
    counts = {}
    for raw in text.lower().split():
        w = raw.strip(".,!?;:\"'()")
        if w:
            counts[w] = counts.get(w, 0) + 1
    return counts

def top_words(text, k):
    items = sorted(word_counts(text).items(), key=lambda p: (-p[1], p[0]))
    return items[:k]

=== Gradebook from CSV Text

--- tests
> data = "name,hw1,hw2,exam\nAna,90,85,88\nBen,70,,65\n\nCal,100,100,100\n"
parse_gradebook(data)["Ana"] == [90, 85, 88]

--- hint
Split the text into lines with text.strip().split("\n") and skip the first (header) line. For each line, split on "," to get the name and the score strings. Turn each score into int(s) if s.strip() isn't empty, otherwise 0.
--- solution
def parse_gradebook(csv_text):
    book = {}
    lines = csv_text.strip().split("\n")
    for line in lines[1:]:
        if not line.strip():
            continue
        parts = line.split(",")
        name = parts[0].strip()
        scores = [int(s) if s.strip() else 0 for s in parts[1:]]
        book[name] = scores
    return book

def class_report(csv_text):
    report = {}
    for name, scores in parse_gradebook(csv_text).items():
        report[name] = round(sum(scores) / len(scores), 1) if scores else 0.0
    return report

--- tests

are_anagrams("abc", "abd") == False
are_anagrams("aab", "abb") == False
group_anagrams(["eat", "tea", "tan", "ate", "nat", "bat"]) == [["ate", "eat", "tea"], ["bat"], ["nat", "tan"]]

--- hint
The signature of a word is its letters in sorted order: "".join(sorted(word)). Every anagram has the same signature. Use a dictionary from signature to a list of words.
--- solution
def are_anagrams(a, b):
    clean = lambda s: sorted(s.replace(" ", "").lower())
    return clean(a) == clean(b)

def group_anagrams(words):
    groups = {}
    for w in words:
        key = "".join(sorted(w))
        groups.setdefault(key, []).append(w)
    return sorted(sorted(g) for g in groups.values())

class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
        self.log = []

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("deposit must be positive")
        self.balance += amount
        self.log.append("deposit " + str(amount))

    def withdraw(self, amount):
        if amount <= 0 or amount > self.balance:
            raise ValueError("invalid withdrawal")
        self.balance -= amount
        self.log.append("withdraw " + str(amount))

    def get_balance(self):
        return self.balance

    def history(self):
        return list(self.log)


def transfer(source, target, amount):
    source.withdraw(amount)
    target.deposit(amount)
