--- hint
Find S, then BFS with a deque of (row, col) and a dict "came_from" that maps each visited cell to the cell you reached it from (S maps to None). The distance is found the moment you pop E. To draw the path, follow came_from back from E to S and mark each cell in between. Convert rows to lists of characters so you can change them, then join them back into strings.
--- solution
from collections import deque

def _bfs(maze):
    rows, cols = len(maze), len(maze[0])
    start = end = None
    for r in range(rows):
        for c in range(cols):
            if maze[r][c] == "S":
                start = (r, c)
            elif maze[r][c] == "E":
                end = (r, c)
    came = {start: None}
    q = deque([start])
    while q:
        cur = q.popleft()
        if cur == end:
            return came, end
        r, c = cur
        for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if 0 <= nr < rows and 0 <= nc < cols and maze[nr][nc] != "#" and (nr, nc) not in came:
                came[(nr, nc)] = cur
                q.append((nr, nc))
    return came, None

def shortest_path_length(maze):
    came, end = _bfs(maze)
    if end is None:
        return -1
    steps, cur = 0, end
    while came[cur] is not None:
        cur = came[cur]
        steps += 1
    return steps

def solve(maze):
    came, end = _bfs(maze)
    if end is None:
        return None
    grid = [list(row) for row in maze]
    cur = came[end]
    while cur is not None and came[cur] is not None:
        grid[cur[0]][cur[1]] = "*"
        cur = came[cur]
    return ["".join(row) for row in grid]

--- hint
Keep a heap of (distance, node), starting with (0, start), and a dict of best distances. Pop the smallest; if you've already finalized that node, skip it. Otherwise record it, then push (d + w, neighbor) for each road. For route, also store prev[neighbor] = node whenever you find a better distance, then walk prev backward from the goal.
--- solution
import heapq

def _run(graph, start):
    dist, prev = {}, {start: None}
    best = {start: 0}
    heap = [(0, start)]
    while heap:
        d, u = heapq.heappop(heap)
        if u in dist:
            continue
        dist[u] = d
        for v, w in graph.get(u, {}).items():
            nd = d + w
            if v not in dist and nd < best.get(v, float("inf")):
                best[v] = nd
                prev[v] = u
                heapq.heappush(heap, (nd, v))
    return dist, prev

def dijkstra(graph, start):
    return _run(graph, start)[0]

def route(graph, start, goal):
    dist, prev = _run(graph, start)
    if goal not in dist:
        return (None, [])
    path, cur = [], goal
    while cur is not None:
        path.append(cur)
        cur = prev[cur]
    return (dist[goal], path[::-1])

--- hint
Kahn's algorithm: count each course's unmet prerequisites (its "in-degree") and build a reverse map from each course to the courses that need it. Start with every course whose count is 0. Each time you "take" a course, decrease the count of everything that needs it; any that hit 0 become available. If you finish with courses left over, there's a cycle. For semesters, process the available courses one whole batch (level) at a time.
--- solution
def _graph(prereqs):
    courses = set(prereqs)
    for pres in prereqs.values():
        courses.update(pres)
    need = {c: 0 for c in courses}
    unlocks = {c: [] for c in courses}
    for c, pres in prereqs.items():
        for p in set(pres):
            need[c] += 1
            unlocks[p].append(c)
    return courses, need, unlocks

def semesters(prereqs):
    courses, need, unlocks = _graph(prereqs)
    level = sorted(c for c in courses if need[c] == 0)
    plan, done = [], 0
    while level:
        plan.append(level)
        done += len(level)
        nxt = []
        for c in level:
            for d in unlocks[c]:
                need[d] -= 1
                if need[d] == 0:
                    nxt.append(d)
        level = sorted(nxt)
    return plan if done == len(courses) else None

def course_order(prereqs):
    plan = semesters(prereqs)
    return None if plan is None else [c for sem in plan for c in sem]

--- hint
Edit distance: d[i][0] = i and d[0][j] = j. Then d[i][j] = d[i-1][j-1] if a[i-1] == b[j-1], otherwise 1 + min(d[i-1][j], d[i][j-1], d[i-1][j-1]). LCS: L[i][j] = L[i-1][j-1] + 1 on a match, otherwise max(L[i-1][j], L[i][j-1]). To get the actual string, start at the bottom-right corner and walk backward: on a match, take the letter and go diagonally; otherwise step toward the bigger neighbor.
--- solution
def edit_distance(a, b):
    m, n = len(a), len(b)
    d = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        d[i][0] = i
    for j in range(n + 1):
        d[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                d[i][j] = d[i - 1][j - 1]
            else:
                d[i][j] = 1 + min(d[i - 1][j], d[i][j - 1], d[i - 1][j - 1])
    return d[m][n]

def lcs(a, b):
    m, n = len(a), len(b)
    L = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])
    out, i, j = [], m, n
    while i > 0 and j > 0:
        if a[i - 1] == b[j - 1]:
            out.append(a[i - 1]); i -= 1; j -= 1
        elif L[i - 1][j] >= L[i][j - 1]:
            i -= 1
        else:
            j -= 1
    return "".join(reversed(out))

def is_valid_board(board):
    for i in range(9):
        row = [ch for ch in board[i] if ch != "."]
        col = [board[r][i] for r in range(9) if board[r][i] != "."]
        br, bc = 3 * (i // 3), 3 * (i % 3)
        box = [board[r][c] for r in range(br, br + 3) for c in range(bc, bc + 3) if board[r][c] != "."]
        for group in (row, col, box):
            if len(group) != len(set(group)):
                return False
    return True

def solve_sudoku(board):
    if not is_valid_board(board):
        return None
    g = [list(row) for row in board]

    def ok(r, c, d):
        if d in g[r]:
            return False
        if any(g[i][c] == d for i in range(9)):
            return False
        br, bc = 3 * (r // 3), 3 * (c // 3)
        return all(g[i][j] != d for i in range(br, br + 3) for j in range(bc, bc + 3))

    def bt():
        for r in range(9):
            for c in range(9):
                if g[r][c] == ".":
                    for d in "123456789":
                        if ok(r, c, d):
                            g[r][c] = d
                            if bt():
                                return True
                            g[r][c] = "."
                    return False
        return True

    return ["".join(row) for row in g] if bt() else None

Tokenize into numbers and single-character symbols. def tokenize(expr):
    tokens, i = [], 0
    while i < len(expr):
        ch = expr[i]
        if ch.isspace():
            i += 1
        elif ch.isdigit() or ch == ".":
            j = i
            while j < len(expr) and (expr[j].isdigit() or expr[j] == "."):
                j += 1
            try:
                tokens.append(float(expr[i:j]))
            except ValueError:
                raise ValueError("bad number " + expr[i:j])
            i = j
        elif ch in "+-*/()":
            tokens.append(ch)
            i += 1
        else:
            raise ValueError("unexpected character " + ch)
    return tokens

def evaluate(expr):
    toks = tokenize(expr)
    pos = [0]

    def peek():
        return toks[pos[0]] if pos[0] < len(toks) else None

    def take():
        t = peek()
        if t is None:
            raise ValueError("unexpected end")
        pos[0] += 1
        return t

    def expression():
        v = term()
        while peek() in ("+", "-"):
            v = v + term() if take() == "+" else v - term()
        return v

    def term():
        v = factor()
        while peek() in ("*", "/"):
            if take() == "*":
                v = v * factor()
            else:
                d = factor()
                if d == 0:
                    raise ZeroDivisionError("division by zero")
                v = v / d
        return v

    def factor():
        t = take()
        if t == "-":
            return -factor()
        if t == "(":
            v = expression()
            if take() != ")":
                raise ValueError("expected )")
            return v
        if isinstance(t, float):
            return t
        raise ValueError("unexpected " + str(t))

    result = expression()
    if peek() is not None:
        raise ValueError("extra input")
    return int(result) if result == int(result) else result
