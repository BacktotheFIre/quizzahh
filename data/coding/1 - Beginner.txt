=== FizzBuzz List
Class: CS 1 - Intro

--- starter

    # your code here
    return result
--- tests
fizzbuzz(1) == ["1"]
fizzbuzz(5) == ["1", "2", "Fizz", "4", "Buzz"]
fizzbuzz(15)[-1] == "FizzBuzz"
fizzbuzz(15)[8] == "Fizz"
fizzbuzz(15)[9] == "Buzz"
len(fizzbuzz(100)) == 100
fizzbuzz(100).count("FizzBuzz") == 6
fizzbuzz(0) == []
--- hint
Loop with for i in range(1, n + 1). Check "divisible by 15" FIRST, because 15 is also divisible by 3 and by 5. Use str(i) to turn a number into a string.
--- solution
def fizzbuzz(n):
    result = []
    for i in range(1, n + 1):
        if i % 15 == 0:
            result.append("FizzBuzz")
        elif i % 3 == 0:
            result.append("Fizz")
        elif i % 5 == 0:
            result.append("Buzz")
        else:
            result.append(str(i))
    return result

--- tests

smallest([]) is None

--- hint
For total, start a variable at 0 and add each number in a loop. For smallest, start with the FIRST item (not 0, because the list might be all negative or all positive) and replace it whenever you find something smaller. average can call your total function.
--- solution
def total(nums):
    t = 0
    for n in nums:
        t += n
    return t

def smallest(nums):
    if len(nums) == 0:
        return None
    best = nums[0]
    for n in nums:
        if n < best:
            best = n
    return best

def average(nums):
    if len(nums) == 0:
        return 0.0
    return total(nums) / len(nums)

    is_palindrome("A man, a plan, a canal: Panama!")  ->  True

--- tests
is_palindrome("racecar") == True
is_palindrome("Racecar") == True
is_palindrome("A man, a plan, a canal: Panama!") == True
is_palindrome("hello") == False
is_palindrome("") == True
is_palindrome("No 'x' in Nixon") == True
is_palindrome("12321") == True
is_palindrome("12345") == False
is_palindrome("ab") == False
--- hint
First build a cleaned-up string: loop over the characters, keep only the ones where ch.isalnum() is True, and make them lowercase. Then compare the cleaned string with its reverse, cleaned[::-1].
--- solution
def is_palindrome(text):
    cleaned = ""
    for ch in text:
        if ch.isalnum():
            cleaned += ch.lower()
    return cleaned == cleaned[::-1]

--- solution
def c_to_f(c):
    return c * 9 / 5 + 32

def f_to_c(f):
    return (f - 32) * 5 / 9

def table(start, stop, step):
    rows = []
    c = start
    while c <= stop:
        rows.append((c, round(c_to_f(c), 1)))
        c += step
    return rows

--- tests

judge(10, [10]) == ["correct"]
judge(5, []) == ["out of guesses"]
judge(50, [75, 25, 50]) == ["too high", "too low", "correct"]

--- hint

--- solution
def judge(secret, guesses):
    out = []
    for g in guesses:
        if g < secret:
            out.append("too low")
        elif g > secret:
            out.append("too high")
        else:
            out.append("correct")
            return out
    out.append("out of guesses")
    return out

--- hint
ord(ch) gives a character's number and chr(n) turns a number back into a character. For a lowercase letter, find its position 0-25 with ord(ch) - ord("a"), add the shift, take % 26 to wrap around, and convert back. Decrypting is just encrypting with -shift.
--- solution
def encrypt(text, shift):
    out = ""
    for ch in text:
        if "a" <= ch <= "z":
            out += chr((ord(ch) - ord("a") + shift) % 26 + ord("a"))
        elif "A" <= ch <= "Z":
            out += chr((ord(ch) - ord("A") + shift) % 26 + ord("A"))
        else:
            out += ch
    return out

def decrypt(text, shift):
    return encrypt(text, -shift)
