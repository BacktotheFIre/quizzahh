--- hint
For balanced: push every opening bracket. When you see a closing bracket, the stack must not be empty, and the popped bracket must be its matching opener (a dict like {")": "(", "]": "[", "}": "{"} helps). At the end, the stack must be empty.
--- solution
class Stack:
    def __init__(self):
        self.items = []

    def push(self, x):
        self.items.append(x)

    def pop(self):
        if not self.items:
            raise IndexError("pop from empty stack")
        return self.items.pop()

    def peek(self):
        if not self.items:
            raise IndexError("peek at empty stack")
        return self.items[-1]

    def is_empty(self):
        return len(self.items) == 0

    def size(self):
        return len(self.items)


def balanced(text):
    pairs = {")": "(", "]": "[", "}": "{"}
    s = Stack()
    for ch in text:
        if ch in "([{":
            s.push(ch)
        elif ch in pairs:
            if s.is_empty() or s.pop() != pairs[ch]:
                return False
    return s.is_empty()

--- hint
push_back has to walk to the last node (or you can keep a tail pointer). For remove, handle the head separately, then walk with a "previous" node so you can do prev.next = cur.next. For reverse, walk the list with three variables, prev, cur and nxt, flipping each cur.next to point at prev.
--- solution
class Node:
    def __init__(self, value, next=None):
        self.value = value
        self.next = next


class LinkedList:
    def __init__(self):
        self.head = None
        self.length = 0

    def __len__(self):
        return self.length

    def push_front(self, x):
        self.head = Node(x, self.head)
        self.length += 1

    def push_back(self, x):
        node = Node(x)
        if self.head is None:
            self.head = node
        else:
            cur = self.head
            while cur.next:
                cur = cur.next
            cur.next = node
        self.length += 1

    def pop_front(self):
        if self.head is None:
            raise IndexError("empty list")
        v = self.head.value
        self.head = self.head.next
        self.length -= 1
        return v

    def find(self, x):
        i, cur = 0, self.head
        while cur:
            if cur.value == x:
                return i
            cur, i = cur.next, i + 1
        return -1

    def remove(self, x):
        prev, cur = None, self.head
        while cur:
            if cur.value == x:
                if prev is None:
                    self.head = cur.next
                else:
                    prev.next = cur.next
                self.length -= 1
                return True
            prev, cur = cur, cur.next
        return False

    def reverse(self):
        prev, cur = None, self.head
        while cur:
            nxt = cur.next
            cur.next = prev
            prev, cur = cur, nxt
        self.head = prev

    def to_list(self):
        out, cur = [], self.head
        while cur:
            out.append(cur.value)
            cur = cur.next
        return out

--- hint
_bucket(key) returns self.buckets[hash(key) % len(self.buckets)]. In put, loop over the bucket's pairs: if a pair has the key, update pair[1] and return; otherwise append [key, value] and add 1 to count, then check whether count / len(buckets) > 0.75. To resize, save all pairs, make a new list of twice as many empty buckets, reset count to 0, and put() every pair again.
--- solution
class HashMap:
    def __init__(self, capacity=8):
        self.buckets = [[] for _ in range(capacity)]
        self.count = 0

    def _bucket(self, key):
        return self.buckets[hash(key) % len(self.buckets)]

    def put(self, key, value):
        b = self._bucket(key)
        for pair in b:
            if pair[0] == key:
                pair[1] = value
                return
        b.append([key, value])
        self.count += 1
        if self.count / len(self.buckets) > 0.75:
            self._grow()

    def _grow(self):
        old = [p for b in self.buckets for p in b]
        self.buckets = [[] for _ in range(len(self.buckets) * 2)]
        self.count = 0
        for k, v in old:
            self.put(k, v)

    def get(self, key, default=None):
        for k, v in self._bucket(key):
            if k == key:
                return v
        return default

    def remove(self, key):
        b = self._bucket(key)
        for i, pair in enumerate(b):
            if pair[0] == key:
                del b[i]
                self.count -= 1
                return True
        return False

    def __len__(self):
        return self.count

    def keys(self):
        return [p[0] for b in self.buckets for p in b]

    def capacity(self):
        return len(self.buckets)

--- hint
Sift up: while i > 0 and a[i] < a[parent], swap them and set i = parent. Sift down: loop forever; find the smallest of i and its (up to two) children; if that's i itself, stop; otherwise swap and continue from the child. In pop, remember a[0], move a.pop() into a[0] (only if anything is left), then sift down.
--- solution
class MinHeap:
    def __init__(self):
        self.a = []

    def __len__(self):
        return len(self.a)

    def peek(self):
        if not self.a:
            raise IndexError("empty heap")
        return self.a[0]

    def push(self, x):
        a = self.a
        a.append(x)
        i = len(a) - 1
        while i > 0:
            p = (i - 1) // 2
            if a[i] < a[p]:
                a[i], a[p] = a[p], a[i]
                i = p
            else:
                break

    def pop(self):
        a = self.a
        if not a:
            raise IndexError("empty heap")
        top = a[0]
        last = a.pop()
        if a:
            a[0] = last
            i = 0
            while True:
                small = i
                for c in (2 * i + 1, 2 * i + 2):
                    if c < len(a) and a[c] < a[small]:
                        small = c
                if small == i:
                    break
                a[i], a[small] = a[small], a[i]
                i = small
        return top


def heap_sort(nums):
    h = MinHeap()
    for x in nums:
        h.push(x)
    return [h.pop() for _ in range(len(nums))]

Recursive helper functions that take a node make this much easier: _insert(node, x) returns the (possibly new) node, and so does _delete(node, x), which lets you write node.left = self._delete(node.left, x). For height: an empty node is -1, otherwise 1 + max(height(left), height(right)).
--- solution
class TreeNode:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None


class BST:
    def __init__(self):
        self.root = None

    def insert(self, x):
        self.root = self._insert(self.root, x)

    def _insert(self, node, x):
        if node is None:
            return TreeNode(x)
        if x < node.value:
            node.left = self._insert(node.left, x)
        elif x > node.value:
            node.right = self._insert(node.right, x)
        return node

    def contains(self, x):
        node = self.root
        while node:
            if x == node.value:
                return True
            node = node.left if x < node.value else node.right
        return False

    def inorder(self):
        out = []
        def walk(n):
            if n:
                walk(n.left)
                out.append(n.value)
                walk(n.right)
        walk(self.root)
        return out

    def height(self):
        def h(n):
            return -1 if n is None else 1 + max(h(n.left), h(n.right))
        return h(self.root)

    def min(self):
        if self.root is None:
            raise ValueError("empty tree")
        n = self.root
        while n.left:
            n = n.left
        return n.value

    def max(self):
        if self.root is None:
            raise ValueError("empty tree")
        n = self.root
        while n.right:
            n = n.right
        return n.value

    def delete(self, x):
        self.root = self._delete(self.root, x)

    def _delete(self, node, x):
        if node is None:
            return None
        if x < node.value:
            node.left = self._delete(node.left, x)
        elif x > node.value:
            node.right = self._delete(node.right, x)
        else:
            if node.left is None:
                return node.right
            if node.right is None:
                return node.left
            succ = node.right
            while succ.left:
                succ = succ.left
            node.value = succ.value
            node.right = self._delete(node.right, succ.value)
        return node

--- hint
Copy first with list(nums). For merge, walk two indexes through the two sorted halves, always taking the smaller front item. For inversions: while merging, whenever you take an item from the RIGHT half, every item still waiting in the left half is bigger than it, so add len(left) - i to the count. For quick sort on sorted input, choose a random pivot (or the middle) so it doesn't hit O(n²).
--- solution
import random

def insertion_sort(nums):
    a = list(nums)
    for i in range(1, len(a)):
        x, j = a[i], i - 1
        while j >= 0 and a[j] > x:
            a[j + 1] = a[j]
            j -= 1
        a[j + 1] = x
    return a

def _merge_count(a):
    if len(a) <= 1:
        return list(a), 0
    mid = len(a) // 2
    left, inv_l = _merge_count(a[:mid])
    right, inv_r = _merge_count(a[mid:])
    out, i, j, inv = [], 0, 0, inv_l + inv_r
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i]); i += 1
        else:
            out.append(right[j]); j += 1
            inv += len(left) - i
    out.extend(left[i:]); out.extend(right[j:])
    return out, inv

def merge_sort(nums):
    return _merge_count(list(nums))[0]

def quick_sort(nums):
    if len(nums) <= 1:
        return list(nums)
    p = nums[random.randrange(len(nums))]
    return quick_sort([x for x in nums if x < p]) + [x for x in nums if x == p] + quick_sort([x for x in nums if x > p])

def count_inversions(nums):
    return _merge_count(list(nums))[1]
